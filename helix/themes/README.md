# 自定义主题放这个目录,文件名即主题名(:theme mybase16)
#
# 想要 base16 的其他口味(gruvbox/onedark/nord 等 16 色),
# 可以从 https://github.com/tinted-theming/tinted-helix 取对应 toml 放进来,
# 例如:
#   curl -o base16-gruvbox.toml \
#     https://raw.githubusercontent.com/tinted-theming/tinted-helix/main/themes/base16-gruvbox.toml
#
# 也可以继承改某几个颜色:
# inherits = "base16_default_dark"
# [palette]
# base0c = "#0xABCDEF"   # 覆盖某个槽位
